<script>
    layui.use(['form', 'element'], function(){
        var form = layui.form;
        var element = layui.element;
        var $ = layui.$;

        // 自定义验证规则
        form.verify({
            pass: [
                /^[\S]{6,12}$/,
                '密码必须6到12位，且不能出现空格'
            ],
            confirmPass: function(value){
                var password = $('#regPassword').val();
                if(value !== password){
                    return '两次输入的密码不一致';
                }
            }
        });

        // 监听提交
        form.on('submit(loginForm)', function(data){
            return true; // 表单正常提交
        });

        // 如果存在错误消息，自动显示注册标签页
        var errorMsg = '${msg}';
        if(errorMsg && errorMsg.indexOf('注册') !== -1) {
            element.tabChange('user', '1'); // 切换到注册标签页
        }
    });
</script>
